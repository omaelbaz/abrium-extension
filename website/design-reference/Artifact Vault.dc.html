import time
from collections import deque

class SlidingWindow:
    """Token-bucket limiter with a
    sliding one-minute window."""

    def __init__(self, limit=60):
        self.limit = limit
        self.hits = deque()

    def allow(self, now=None):
        now = now or time.time()
        while self.hits and \
              now - self.hits[0] > 60:
            self.hits.popleft()
        if len(self.hits) >= self.limit:
            return False
        self.hits.append(now)
        return True
